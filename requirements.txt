# MinbarLive - Dependencies
# Versions pinned for reproducibility. Python 3.11+ — numpy and scipy below
# both declare Requires-Python >=3.11, so 3.10 cannot install this file at all.
# CI and the release builds run 3.12.

# OpenAI API client
openai==2.14.0

# Gemini API client (optional provider; imported lazily)
google-genai==2.10.0

# Anthropic Claude API client (optional provider; imported lazily)
anthropic==0.116.0

# Deepgram streaming transcription (optional, real-time mode only; imported lazily)
deepgram-sdk==7.4.0

# WebSocket transport for the streaming engines — the DEFAULT (OpenAI Realtime)
# plus Deepgram and Gemini Live. openai only pulls it in via its [realtime]
# extra, and the other two only require >=12, so pin it explicitly here or a
# fresh install could resolve a version the default engine can't use.
websockets==16.0

# Load OPENAI_API_KEY from .env
python-dotenv==1.2.1

# Audio processing
sounddevice==0.5.3
scipy==1.16.3
numpy==2.4.0

# Loopback audio capture from output devices (speakers/headphones) on Windows
SoundCard>=0.4.6

# Voice-activity noise filter (webrtcvad fork with prebuilt wheels; the app
# degrades to pass-through if it is missing)
webrtcvad-wheels==2.0.14

# GUI
screeninfo==0.8.1

# PySide6-Essentials, NOT PySide6: the full package pulls PySide6-Addons, and a
# full install measures 634 MB on disk with Qt6WebEngineCore.dll alone at 195 MB
# (an entire Chromium we never use). Essentials carries QtCore, QtGui and
# QtWidgets, which is all gui/ imports — QML/QtQuick was deliberately rejected,
# see the module docstring in gui/subtitle_window.py.
#
# MinbarLive is AGPL-3.0, so Qt's LGPL imposes no practical constraint beyond
# shipping the Qt LGPL notice.
PySide6-Essentials==6.11.1

# Header logo rendering (crop + high-quality resize).
Pillow==12.3.0

# No arabic-reshaper or python-bidi: Qt shapes and bidi-orders text with
# HarfBuzz on every platform, which is one of the three bug classes the
# migration off CustomTkinter removed. Do not add them back.

# Secure API key storage (OS keychain)
keyring==25.6.0

# Testing (development only)
pytest==9.0.2

# Linting (development only)
ruff==0.11.12

# Creating Exe
pyinstaller==6.18.0
pyinstaller-hooks-contrib==2025.11
